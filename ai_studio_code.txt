import React, { useState } from 'react';
import { Sparkles, Send, MessageCircle, Bot, HelpCircle, Loader2 } from 'lucide-react';
import { ASTROLOGER_INFO } from '../data/astrologyData';
import { getGeneralWhatsAppUrl } from '../utils/calendar';

interface AskGeminiSectionProps {
  onOpenBooking?: () => void;
  userProfile?: {
    dob?: string;
    tob?: string;
    pob?: string;
    preferredSign?: string;
  };
}

export const AskGeminiSection: React.FC<AskGeminiSectionProps> = ({ onOpenBooking, userProfile }) => {
  const [question, setQuestion] = useState('');
  const [answer, setAnswer] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [activePreset, setActivePreset] = useState<string | null>(null);

  const sampleQuestions = [
    'Which Vedic gemstone is most auspicious for my current transit?',
    'How do Rahu and Ketu karmic nodal shifts affect career stability?',
    'What are the key factors checked in Ashtakoota Kundali Milan?',
    'Which simple daily mantra activates Jupiter (Guru) blessings?',
    'What should I do during my Sade Sati or Saturn return period?'
  ];

  const handleAskQuestion = async (customQuestion?: string) => {
    const q = customQuestion || question;
    if (!q.trim()) return;

    setIsLoading(true);
    setAnswer(null);

    try {
      const res = await fetch('/api/ask-gemini', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          question: q.trim(),
          birthDetails: userProfile ? {
            dob: userProfile.dob,
            tob: userProfile.tob,
            pob: userProfile.pob,
            sign: userProfile.preferredSign
          } : undefined
        })
      });

      const data = await res.json();
      setAnswer(data.answer || 'Under classical Vedic principles, positive intentions and sattvic disciplines harmonize planetary transits.');
    } catch (err) {
      console.error('Error querying Google Gemini Free AI:', err);
      setAnswer('Under classical Parashari Jyotish principles, mindful reflection and sattvic routines harmonize planetary transits. For personal guidance, consult Astrologer Ayan Ghosh on WhatsApp (+91 6294601364).');
    } finally {
      setIsLoading(false);
    }
  };

  const handleSelectPreset = (q: string) => {
    setActivePreset(q);
    setQuestion(q);
    handleAskQuestion(q);
  };

  return (
    <section id="ask-gemini" className="py-16 lg:py-24 bg-[#070810] border-b border-amber-500/15 relative overflow-hidden">
      <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[600px] h-[600px] bg-gradient-to-tr from-amber-500/10 via-purple-900/15 to-blue-900/10 rounded-full blur-3xl pointer-events-none" />

      <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 relative">
        <div className="text-center max-w-2xl mx-auto mb-10 space-y-3">
          <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-gradient-to-r from-blue-500/20 via-purple-500/20 to-amber-500/20 border border-blue-400/30 text-blue-300 text-xs font-medium">
            <Bot className="w-3.5 h-3.5 text-blue-400" />
            <span>Powered by Google Gemini Free AI</span>
            <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
          </div>

          <h2 className="font-cinzel text-3xl sm:text-4xl font-bold text-white text-balance">
            Ask Astrological Questions with Google Gemini Free AI
          </h2>

          <p className="text-sm text-zinc-300 font-light">
            Instant, intelligent Vedic astrological insights and remedial wisdom 24/7. Ask about career timing, relationship compatibility, gemstones, mantras, and planetary transits.
          </p>
        </div>

        <div className="rounded-2xl border border-amber-500/30 bg-[#0a0c18]/90 backdrop-blur-xl p-6 sm:p-8 shadow-2xl space-y-6">
          <div>
            <div className="text-xs uppercase tracking-wider text-amber-400 font-semibold mb-2.5 flex items-center gap-1.5">
              <HelpCircle className="w-3.5 h-3.5 text-amber-400" />
              <span>Suggested Astrological Inquiries:</span>
            </div>
            <div className="flex flex-wrap gap-2">
              {sampleQuestions.map((q, idx) => (
                <button
                  key={idx}
                  onClick={() => handleSelectPreset(q)}
                  className={`text-xs px-3 py-1.5 rounded-lg border transition-all text-left cursor-pointer ${
                    activePreset === q
                      ? 'border-amber-400 bg-amber-400/20 text-amber-200'
                      : 'border-zinc-800 bg-zinc-900/60 hover:border-amber-400/40 text-zinc-300 hover:text-white'
                  }`}
                >
                  {q}
                </button>
              ))}
            </div>
          </div>

          <form
            onSubmit={(e) => {
              e.preventDefault();
              handleAskQuestion();
            }}
            className="flex flex-col sm:flex-row items-stretch gap-3"
          >
            <div className="relative flex-1">
              <input
                type="text"
                required
                placeholder="Ask Google Gemini Free AI any astrology question (e.g. Which gemstone suits me?)..."
                value={question}
                onChange={(e) => setQuestion(e.target.value)}
                className="w-full px-4 py-3 text-xs sm:text-sm rounded-xl bg-zinc-900/90 border border-zinc-700 text-white placeholder-zinc-500 focus:border-amber-400 focus:outline-none pr-10"
              />
              <Sparkles className="absolute right-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-amber-400/60 pointer-events-none" />
            </div>

            <button
              type="submit"
              disabled={isLoading || !question.trim()}
              className="px-6 py-3 rounded-xl bg-gradient-to-r from-amber-400 via-amber-500 to-yellow-500 hover:from-amber-300 hover:to-yellow-400 disabled:opacity-50 text-black font-semibold text-xs uppercase tracking-wider transition-all flex items-center justify-center gap-2 cursor-pointer shadow-lg shrink-0"
            >
              {isLoading ? (
                <>
                  <Loader2 className="w-4 h-4 animate-spin text-black" />
                  <span>Consulting Gemini AI...</span>
                </>
              ) : (
                <>
                  <Send className="w-4 h-4 text-black" />
                  <span>Ask Free AI</span>
                </>
              )}
            </button>
          </form>

          {answer && (
            <div className="p-5 sm:p-6 rounded-xl bg-gradient-to-br from-[#0e1224] to-[#080a14] border border-amber-500/30 space-y-4 animate-fadeIn">
              <div className="flex items-center justify-between pb-3 border-b border-zinc-800">
                <div className="flex items-center gap-2">
                  <div className="w-6 h-6 rounded-md bg-blue-500/20 border border-blue-400/40 flex items-center justify-center text-blue-300 text-xs font-bold font-mono">
                    G
                  </div>
                  <div>
                    <span className="font-cinzel text-xs font-bold text-white block">
                      Google Gemini Free AI Astrological Insight
                    </span>
                    <span className="text-[10px] text-emerald-400 font-mono block">
                      Vedic Parashari & Jaimini Grounded
                    </span>
                  </div>
                </div>

                <span className="text-[10px] px-2 py-0.5 rounded bg-amber-400/20 text-amber-300 font-mono">
                  100% Free Access
                </span>
              </div>

              <div className="text-xs sm:text-sm text-zinc-200 leading-relaxed font-light whitespace-pre-line">
                {answer}
              </div>

              <div className="pt-3 border-t border-zinc-800/80 flex flex-col sm:flex-row sm:items-center justify-between gap-3 text-xs">
                <span className="text-zinc-400">
                  Want an in-depth 1-on-1 private reading with full natal chart Dossier?
                </span>

                <div className="flex items-center gap-3">
                  <a
                    href={getGeneralWhatsAppUrl(`Namaskar Astrologer Ayan Ghosh ji, I asked Google Gemini AI: "${question}". I would like to book a personal reading.`)}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="text-emerald-400 hover:text-emerald-300 font-medium flex items-center gap-1.5"
                  >
                    <MessageCircle className="w-3.5 h-3.5 text-emerald-400" />
                    <span>WhatsApp Ayan (+91 {ASTROLOGER_INFO.phone})</span>
                  </a>

                  {onOpenBooking && (
                    <button
                      onClick={onOpenBooking}
                      className="px-3 py-1.5 rounded-lg bg-amber-400 hover:bg-amber-300 text-black font-semibold text-xs transition-colors cursor-pointer"
                    >
                      Book Reading
                    </button>
                  )}
                </div>
              </div>
            </div>
          )}
        </div>
      </div>
    </section>
  );
};
